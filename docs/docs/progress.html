<!-- JS helper: set any % -->
<script>
function setRingProgress(ringEl, pct) {
  const CIRC = 326.73;
  const circle = ringEl.querySelector('.wui-progress-ring-fill');
  circle.style.strokeDashoffset = CIRC * (1 - pct / 100);
  ringEl.querySelector('.wui-progress-ring-pct').textContent = pct + '%';
}
</script>

<!-- JS: set segments dynamically -->
<script>
function setSegmentedRing(ringEl, segments) {
  // segments: [{ pct, color }, …] — pcts must sum to 100
  const CIRC = 326.73;
  let cum = 0;
  ringEl.querySelectorAll('.wui-progress-ring-segment').forEach(function(el, i) {
    const s = segments[i]; if (!s) return;
    el.setAttribute('stroke-dasharray', (CIRC * s.pct / 100) + ' ' + CIRC);
    el.setAttribute('transform', 'rotate(' + (-90 + cum * 3.6) + ' 60 60)');
    cum += s.pct;
  });
}
</script>
